    838  class PreLayerNormTransformerEncoderLayer(nn.Module):
    839      """Pre-LN Transformer Encoder Layer (更稳定的训练)"""
    840
    841      def __init__(self, d_model: int, nhead: int, dim_feedforward: int = 2048,
    842                   dropout: float = 0.1, activation: str = "gelu"):

    845          self.self_attn = nn.MultiheadAttention(d_model, nhead, dropout=dropout,
    846                                                  batch_first=True)

    849          self.linear1 = nn.Linear(d_model, dim_feedforward)
    850          self.activation = nn.GELU() if activation == "gelu" else nn.ReLU()
    851          self.dropout_act = nn.Dropout(dropout)
    852          self.linear2 = nn.Linear(dim_feedforward, d_model)
    853          self.dropout2 = nn.Dropout(dropout)

    855      def forward(self, src: torch.Tensor, src_mask: Optional[torch.Tensor] = None,
    856                  src_key_padding_mask: Optional[torch.Tensor] = None) -> torch.Tensor:

    861                                        key_padding_mask=src_key_padding_mask)
    862          src = residual + self.dropout1(attn_out)

    872  class LearnedPositionalEncoding(nn.Module):
    873      """可学习的位置编码"""
    874
    875      def __init__(self, d_model: int, max_len: int = 64):
    876          super().__init__()

    883  class AttentivePooling(nn.Module):
    884      """注意力池化：加权平均序列 → 固定维度"""
    885
    886      def __init__(self, d_model: int):
    887          super().__init__()
    888          self.attn = nn.Sequential(
    889              nn.Linear(d_model, d_model // 2),
    890              nn.Tanh(),
    891              nn.Linear(d_model // 2, 1),
    892          )
    893
    894      def forward(self, x: torch.Tensor, mask: Optional[torch.Tensor] = None) -> torch.Tensor:
    895          """
    896          Args:
    897              x: [B, L, D]
    898              mask: [B, L] True=valid
    899          Returns:
    900              [B, D]
    901          """
    902          scores = self.attn(x).squeeze(-1)  # [B, L]
    903          if mask is not None:
    904              scores = scores.masked_fill(~mask, torch.finfo(scores.dtype).min)
    905          weights = F.softmax(scores, dim=-1).unsqueeze(-1)  # [B, L, 1]
    906          return (x * weights).sum(dim=1)

    913  class SymptomEncoder(nn.Module):
    914      """多层Transformer症状编码器"""
    915
    916      def __init__(self, num_symptoms: int, d_model: int = 512, nhead: int = 8,
    917                   num_layers: int = 6, dim_ff: int = 2048, dropout: float = 0.1):
    918          super().__init__()
    919          self.d_model = d_model
    920          self.embedding = nn.Embedding(num_symptoms + 1, d_model, padding_idx=0)
    921          self.pos_encoder = LearnedPositionalEncoding(d_model)
    922          self.dropout_emb = nn.Dropout(dropout)
    923
    924          self.layers = nn.ModuleList([
    925              PreLayerNormTransformerEncoderLayer(d_model, nhead, dim_ff, dropout)
    926              for _ in range(num_layers)
    927          ])
    928          self.final_norm = nn.LayerNorm(d_model)
    929          self.pooling = AttentivePooling(d_model)
    930
    931          self._init_weights()
    932
    933      def _init_weights(self):
    934          nn.init.normal_(self.embedding.weight, mean=0, std=self.d_model ** -0.5)
    935          nn.init.constant_(self.embedding.weight[0], 0)
    936
    937      def forward(self, symptom_ids: torch.Tensor,
    938                  mask: Optional[torch.Tensor] = None) -> Tuple[torch.Tensor, torch.Tensor]:
    939          """
    940          Args:
    941              symptom_ids: [B, L], 1-based, 0=padding
    942              mask: [B, L], True=valid
    943          Returns:
    944              pooled: [B, D]
    945              sequence: [B, L, D]
    946          """
    947          if mask is None:
    948              mask = symptom_ids != 0
    949
    950          x = self.embedding(symptom_ids) * (self.d_model ** 0.5)
    951          x = self.pos_encoder(x)
    952          x = self.dropout_emb(x)
    953
    954          key_padding_mask = ~mask
    955          for layer in self.layers:
    956              x = layer(x, src_key_padding_mask=key_padding_mask)
    957
    958          x = self.final_norm(x)
    959          pooled = self.pooling(x, mask)
