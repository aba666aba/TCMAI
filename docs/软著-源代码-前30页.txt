    580      def _identify_syndromes(self, symptoms: List[SymptomInfo]) -> List[SyndromeResult]:
    581          """证型判断"""
    582          symptom_names = [s.standardized_name for s in symptoms]
    583          results = []
    584
    585          for syndrome_name, rule in self.syndrome_rules.items():
    586              matched = [s for s in rule["symptoms"] if s in symptom_names]
    587
    588              if not rule["symptoms"]:
    589                  logger.warning("证型 '%s' 的症状列表为空，跳过该证型", syndrome_name)
    590                  continue
    591
    592              confidence = len(matched) / len(rule["symptoms"])
    593
    594              if confidence >= 0.5:
    595                  result = SyndromeResult(
    596                      syndrome_name=syndrome_name,
    597                      confidence=confidence,
    598                      symptoms=matched,
    599                      explanation=rule["explanation"],
    600                  )
    601                  results.append(result)
    602
    603          results.sort(key=lambda x: x.confidence, reverse=True)
    604          logger.info("证型判断完成：%s", [r.syndrome_name for r in results])
    605          return results
